        function parseCSV(csvText) {
            const lines = csvText.trim().split('\n');
            const headers = lines[0].split(',');
            const data = [];
            
            for (let i = 1; i < lines.length; i++) {
                const values = [];
                let current = '';
                let inQuotes = false;
                
                for (let j = 0; j < lines[i].length; j++) {
                    const char = lines[i][j];
                    if (char === '"') {
                        inQuotes = !inQuotes;
                    } else if (char === ',' && !inQuotes) {
                        values.push(current.trim());
                        current = '';
                    } else {
                        current += char;
                    }
                }
                values.push(current.trim());
                
                if (values.length === headers.length) {
                    const row = {};
                    headers.forEach((header, index) => {
                        row[header] = values[index];
                    });
                    data.push(row);
                }
            }
            return data;
        }
        
        function loadSampleData() {
            try {
                questions = parseCSV(sampleData);
                document.getElementById('status').innerHTML = 
                    `<strong>로딩 완료!</strong> ${questions.length}개 문제 로드됨`;
                
                document.getElementById('question-container').style.display = 'block';
                showQuestion(0);
                
            } catch (error) {
                document.getElementById('status').innerHTML = 
                    `<strong>오류:</strong> ${error.message}`;
                console.error('데이터 로딩 오류:', error);
            }
        }
        
        function showQuestion(index) {
            const question = questions[index];
            currentIndex = index;
            
            document.getElementById('question-num').textContent = index + 1;
            document.getElementById('question-code').textContent = question.QCODE;
            document.getElementById('question-type').textContent = question.TYPE;
            document.getElementById('question-text').textContent = question.QUESTION;
            
            const choicesContainer = document.getElementById('choices');
            choicesContainer.innerHTML = '';
            
            if (question.TYPE === '진위형') {
                ['O', 'X'].forEach(choice => {
                    const btn = document.createElement('button');
                    btn.className = 'choice-btn';
                    btn.textContent = choice === 'O' ? '맞음 (O)' : '틀림 (X)';
                    btn.onclick = () => selectAnswer(choice);
                    
                    if (userAnswers[question.QCODE] === choice) {
                        btn.classList.add('selected');
                    }
                    
                    choicesContainer.appendChild(btn);
                });
            } else if (question.TYPE === '선택형') {
                ['1', '2', '3', '4'].forEach(choice => {
                    const btn = document.createElement('button');
                    btn.className = 'choice-btn';
                    btn.textContent = `${choice}번`;
                    btn.onclick = () => selectAnswer(choice);
                    
                    if (userAnswers[question.QCODE] === choice) {
                        btn.classList.add('selected');
                    }
                    
                    choicesContainer.appendChild(btn);
                });
            }
            
            // 버튼 상태 업데이트
            document.getElementById('prev-btn').disabled = index === 0;
            document.getElementById('next-btn').textContent = 
                index === questions.length - 1 ? '결과 보기' : '다음';
        }
        
        function selectAnswer(answer) {
            const question = questions[currentIndex];
            userAnswers[question.QCODE] = answer;
            
            // 선택 상태 업데이트
            document.querySelectorAll('.choice-btn').forEach(btn => {
                btn.classList.remove('selected', 'correct', 'incorrect');
            });
            
            document.querySelectorAll('.choice-btn').forEach(btn => {
                if ((answer === 'O' && btn.textContent.includes('맞음')) ||
                    (answer === 'X' && btn.textContent.includes('틀림')) ||
                    btn.textContent.includes(`${answer}번`)) {
                    btn.classList.add('selected');
                    
                    // 정답 확인
                    if (answer === question.ANSWER) {
                        btn.classList.add('correct');
                    } else {
                        btn.classList.add('incorrect');
                    }
                }
            });
        }
        
        function prevQuestion() {
            if (currentIndex > 0) {
                showQuestion(currentIndex - 1);
            }
        }
        
        function nextQuestion() {
            if (currentIndex < questions.length - 1) {
                showQuestion(currentIndex + 1);
            } else {
                showResults();
            }
        }
        
        function showResults() {
            let correct = 0;
            let total = questions.length;
            
            questions.forEach(q => {
                if (userAnswers[q.QCODE] === q.ANSWER) {
                    correct++;
                }
            });
            
            document.getElementById('status').innerHTML = 
                `<strong>테스트 완료!</strong> ${correct}/${total}문제 정답 (${Math.round(correct/total*100)}%)`;
            
            alert(`샘플 테스트 결과:\n정답: ${correct}/${total}문제\n정답률: ${Math.round(correct/total*100)}%`);
        }
